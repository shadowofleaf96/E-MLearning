function arrayManipulation(n, queries) {
    // Initialize array with zeros
    const arr = new Array(n + 1).fill(0);
    
    // Add values at boundaries
    for (let i = 0; i < queries.length; i++) {
        const [a, b, k] = queries[i];
        arr[a - 1] += k;
        if (b < n) {
            arr[b] -= k;
        }
    }
    
    let max = 0;
    let current = 0;
    
    for (let i = 0; i < n; i++) {
        current += arr[i];
        if (current > max) max = current;
    }
    
    return max;
}
